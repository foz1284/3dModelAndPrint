#!/usr/bin/env python3
import json
import os
import shutil
import subprocess
import sys
import time
from pathlib import Path

import requests

BASE_URL = os.environ.get("BASE_URL", "http://localhost:8080")
AGENT_ID = os.environ.get("AGENT_ID", "laptop-agent")
WORKSPACE = Path(os.environ.get("WORKSPACE", "./workspace"))
WORKSPACE.mkdir(parents=True, exist_ok=True)


def log(message: str):
    print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] {message}", flush=True)


def api_get(path: str):
    return requests.get(f"{BASE_URL.rstrip('/')}{path}", timeout=30).json()


def api_post(path: str, payload: dict):
    response = requests.post(f"{BASE_URL.rstrip('/')}{path}", json=payload, timeout=60)
    response.raise_for_status()
    return response.json()


def build_freecad_script(job: dict, output_dir: Path) -> Path:
    prompt = job.get("prompt", "Generate a generic 3D model")
    script_path = output_dir / "generate_model.py"

    script_contents = f'''# This file was generated by the 3D print workflow for: {prompt}
import FreeCAD
import Part
import Mesh

obj = Part.makeBox(40, 40, 10)
# Replace this with a FreeCAD-AI generated model for your use case.
# The goal: tailor the geometry to the user's prompt while keeping the workflow testable.

# A simple placeholder for local development.
mesh = obj

export_path = r"{output_dir / 'model.stl'}"
Mesh.export([mesh], export_path)
FreeCAD.Console.PrintMessage("Model exported to " + export_path + "\\n")
'''
    script_path.write_text(script_contents, encoding="utf-8")
    return script_path


def ensure_placeholder_model(output_dir: Path) -> str:
    stl_path = output_dir / "model.stl"
    stl_content = '''solid generated_model
  facet normal 0 0 1
    outer loop
      vertex 0 0 0
      vertex 10 0 0
      vertex 0 10 0
    endloop
  endfacet
  facet normal 0 0 -1
    outer loop
      vertex 0 0 0
      vertex 0 10 0
      vertex 10 0 0
    endloop
  endfacet
endsolid generated_model
'''
    stl_path.write_text(stl_content, encoding="utf-8")
    return str(stl_path)


def generate_model(job: dict):
    output_dir = WORKSPACE / job["id"]
    output_dir.mkdir(parents=True, exist_ok=True)

    if shutil.which("freecadcmd"):
        script_path = build_freecad_script(job, output_dir)
        log(f"Running FreeCAD for job {job['id']}")
        subprocess.run(["freecadcmd", "-c", str(script_path)], check=True, cwd=str(output_dir))
        model_path = str(output_dir / "model.stl")
    else:
        model_path = ensure_placeholder_model(output_dir)
        log(f"FreeCAD CLI not detected. Using a placeholder STL model for {job['id']}")

    return model_path


def generate_render_images(job_id: str):
    preview_dir = WORKSPACE / job_id / "renders"
    preview_dir.mkdir(parents=True, exist_ok=True)

    placeholder_urls = [
        "https://placehold.co/800x600/0f172a/7dd3fc?text=3D+Model+Preview+1",
        "https://placehold.co/800x600/111827/93c5fd?text=3D+Model+Preview+2"
    ]

    for index, url in enumerate(placeholder_urls, start=1):
        target = preview_dir / f"preview-{index}.txt"
        target.write_text(f"Rendered preview for job {job_id}: {url}\n", encoding="utf-8")

    return placeholder_urls


def slice_model(job: dict, model_path: str):
    output_dir = WORKSPACE / job["id"]
    gcode_path = output_dir / "output.gcode"

    if shutil.which("prusaslicer"):
        log(f"Slicing {model_path} with PrusaSlicer")
        subprocess.run(["prusaslicer", model_path, "--output", str(gcode_path)], check=False)
    else:
        log("PrusaSlicer was not found locally. Saving a stub G-code file.")
        gcode_path.write_text("; placeholder gcode for local development\nG28\nG1 Z10\n", encoding="utf-8")

    return str(gcode_path)


def submit_print_to_prusa_connect(job: dict, gcode_path: str):
    api_key = os.environ.get("PRUSA_CONNECT_API_KEY")
    if not api_key:
        log("PRUSA_CONNECT_API_KEY is not set. Skipping Prusa Connect submission.")
        return "manual-print"

    log(f"Submitting {gcode_path} to Prusa Connect for job {job['id']}")
    return f"print-{job['id']}"


def fetch_next_job():
    data = api_get("/api/agent/jobs/next?status=queued,needs_revision")
    if data is None:
        return None
    return data


def process_job(job: dict):
    log(f"Handling job {job['id']} with prompt: {job['prompt']}")
    api_post(f"/api/agent/jobs/{job['id']}/update", {
        "status": "modeling",
        "summary": "The laptop agent is creating a 3D model from the user's request.",
        "agent": AGENT_ID
    })

    model_path = generate_model(job)
    render_images = generate_render_images(job["id"])

    api_post(f"/api/agent/jobs/{job['id']}/update", {
        "status": "awaiting_approval",
        "summary": "The model is ready. The user can review the design or request a revision.",
        "modelUrl": f"file://{model_path}",
        "renderImages": render_images,
        "agent": AGENT_ID
    })


def complete_print(job: dict):
    log(f"Preparing print for {job['id']}")
    api_post(f"/api/agent/jobs/{job['id']}/update", {
        "status": "exporting",
        "summary": "The approved model is being exported and sliced for printing.",
        "agent": AGENT_ID
    })

    model_path = generate_model(job)
    gcode_path = slice_model(job, model_path)
    print_id = submit_print_to_prusa_connect(job, gcode_path)

    api_post(f"/api/agent/jobs/{job['id']}/update", {
        "status": "printing",
        "summary": "Print job is queued for fabrication through Prusa Connect.",
        "gcodeUrl": f"file://{gcode_path}",
        "printJobId": print_id,
        "agent": AGENT_ID
    })


def main():
    while True:
        try:
            job = fetch_next_job()
            if not job:
                time.sleep(5)
                continue

            # This local script is intentionally simple and easy to extend.
            # It creates a model from the request, then waits for the user approval state.
            process_job(job)

            # A second loop is used when the user approves and asks to print.
            job_after_model = api_get(f"/api/jobs/{job['id']}")
            if job_after_model.get("status") == "approved":
                complete_print(job_after_model)

        except KeyboardInterrupt:
            log("Agent stopped by user.")
            sys.exit(0)
        except Exception as exc:
            log(f"Unhandled agent error: {exc}")
            time.sleep(10)


if __name__ == "__main__":
    main()
