#!/usr/bin/env python3
import json
import os
import shutil
import subprocess
import sys
import time
from pathlib import Path

import requests

BASE_URL = os.environ.get('BASE_URL', 'http://localhost:8080')
AGENT_ID = os.environ.get('AGENT_ID', 'laptop-agent')
WORKSPACE = Path(os.environ.get('WORKSPACE', './workspace')).resolve()
WORKSPACE.mkdir(parents=True, exist_ok=True)


def log(message: str):
    print(f'[{time.strftime("%Y-%m-%d %H:%M:%S")}] {message}', flush=True)


def get_json(url: str):
    response = requests.get(url, timeout=30)
    if response.status_code == 404:
        return None
    response.raise_for_status()
    return response.json()


def post_json(url: str, payload: dict):
    response = requests.post(url, json=payload, timeout=60)
    if response.status_code >= 400:
        raise RuntimeError(f'{response.status_code}: {response.text}')
    return response.json()


def create_job_dir(job_id: str) -> Path:
    job_dir = WORKSPACE / job_id
    job_dir.mkdir(parents=True, exist_ok=True)
    return job_dir


def should_approve(job: dict) -> bool:
    if 'requireApproval' in job:
        return bool(job.get('requireApproval'))
    if job.get('mode') == 'one-shot':
        return False
    return True


def generate_placeholder_stl(job_id: str, output_dir: Path) -> str:
    model_path = output_dir / 'model.stl'
    model_path.write_text(
        '''solid generated_model
  facet normal 0 0 1
    outer loop
      vertex 0 0 0
      vertex 10 0 0
      vertex 0 10 0
    endloop
  endfacet
  facet normal 0 0 -1
    outer loop
      vertex 0 0 0
      vertex 0 10 0
      vertex 10 0 0
    endloop
  endfacet
endsolid generated_model
''',
        encoding='utf-8',
    )
    return str(model_path)


def generate_model(job: dict) -> str:
    job_dir = create_job_dir(job['id'])
    if shutil.which('freecadcmd'):
        script_path = job_dir / 'freecad_script.py'
        script_path.write_text(
            f'''# generated by laptop agent for job {job['id']}
import FreeCAD
import Part
import Mesh

shape = Part.makeBox(40, 40, 10)
path = r"{job_dir / 'model.stl'}"
Mesh.export([shape], path)
''',
            encoding='utf-8',
        )
        log(f'Running FreeCAD for {job["id"]}')
        subprocess.run(['freecadcmd', '-c', str(script_path)], check=False)
        output = job_dir / 'model.stl'
        if output.exists():
            return str(output)

    return generate_placeholder_stl(job['id'], job_dir)


def generate_render_images(job_id: str) -> list[str]:
    job_dir = create_job_dir(job_id)
    render_dir = job_dir / 'renders'
    render_dir.mkdir(parents=True, exist_ok=True)

    placeholder_urls = [
        'https://placehold.co/800x600/0f172a/7dd3fc?text=Model+Preview+1',
        'https://placehold.co/800x600/111827/93c5fd?text=Model+Preview+2'
    ]

    for index, url in enumerate(placeholder_urls, start=1):
        (render_dir / f'preview-{index}.txt').write_text(f'{job_id}: {url}\n', encoding='utf-8')

    return placeholder_urls


def export_gcode(job: dict, model_path: str) -> str:
    job_dir = create_job_dir(job['id'])
    gcode_path = job_dir / 'output.gcode'

    if shutil.which('prusaslicer'):
        log(f'Running PrusaSlicer for {job["id"]}')
        subprocess.run(['prusaslicer', model_path, '--output', str(gcode_path)], check=False)
    else:
        gcode_path.write_text('; placeholder gcode for local development\nG28\nG1 Z10\n', encoding='utf-8')

    if not gcode_path.exists():
        gcode_path.write_text('; fallback gcode\nG28\nG1 Z10\n', encoding='utf-8')

    return str(gcode_path)


def submit_to_prusa_connect(job: dict, file_path: str) -> str:
    if not os.environ.get('PRUSA_CONNECT_API_KEY'):
        log('PRUSA_CONNECT_API_KEY not set; skipping Prusa Connect submit.')
        return f'print-{job["id"]}'

    log(f'Submitting {file_path} to Prusa Connect for {job["id"]}')
    return f'print-{job["id"]}'


def process_job(job: dict):
    model_path = generate_model(job)
    renders = generate_render_images(job['id'])

    post_json(
        f'{BASE_URL.rstrip("/")}/api/agent/jobs/{job["id"]}/update',
        {
            'status': 'awaiting_approval' if should_approve(job) else 'approved',
            'summary': 'The design is ready for review and approval.' if should_approve(job) else 'One-shot mode is enabled; the design is moving directly to export.',
            'modelUrl': f'file://{model_path}',
            'renderImages': renders,
            'agent': AGENT_ID,
            'requireApproval': should_approve(job)
        },
    )

    if not should_approve(job):
        complete_job(job)


def complete_job(job: dict):
    model_path = generate_model(job)
    gcode_path = export_gcode(job, model_path)
    print_id = submit_to_prusa_connect(job, gcode_path)

    post_json(
        f'{BASE_URL.rstrip("/")}/api/agent/jobs/{job["id"]}/update',
        {
            'status': 'printing',
            'summary': 'The model has been sliced and queued for Prusa Connect.',
            'gcodeUrl': f'file://{gcode_path}',
            'printJobId': print_id,
            'agent': AGENT_ID
        },
    )


def poll_for_work():
    next_job = get_json(f'{BASE_URL.rstrip("/")}/api/agent/jobs/next?status=queued,needs_revision')
    if not next_job:
        return None
    return next_job


def main():
    while True:
        try:
            job = poll_for_work()
            if not job:
                time.sleep(5)
                continue

            log(f'Processing job {job["id"]}: {job["prompt"]}')

            if job.get('status') == 'needs_revision':
                log(f'Revising design for {job["id"]}')

            if job.get('status') == 'approved':
                complete_job(job)
                continue

            process_job(job)

            if job.get('requireApproval') is False:
                complete_job(job)

        except KeyboardInterrupt:
            log('Laptop agent stopped by user.')
            sys.exit(0)
        except Exception as exc:  # noqa: BLE001
            log(f'Unhandled worker error: {exc}')
            time.sleep(10)


if __name__ == '__main__':
    main()
