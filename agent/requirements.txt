#!/usr/bin/env python3
import os
import shutil
import subprocess
import sys
import time
from pathlib import Path

import requests
from azure.storage.blob import BlobServiceClient

BASE_URL = os.environ.get('BASE_URL', 'http://localhost:8080')
AGENT_ID = os.environ.get('AGENT_ID', 'laptop-agent')
AGENT_KEY = os.environ.get('AGENT_API_KEY', 'dev-local-agent-key')
WORKSPACE = Path(os.environ.get('WORKSPACE', './workspace')).resolve()
WORKSPACE.mkdir(parents=True, exist_ok=True)


def log(message: str):
    print(f'[{time.strftime("%Y-%m-%d %H:%M:%S")}] {message}', flush=True)


def get_json(url: str):
    response = requests.get(url, headers={'x-agent-key': AGENT_KEY}, timeout=30)
    if response.status_code == 404:
        return None
    response.raise_for_status()
    return response.json()


def post_json(url: str, payload: dict):
    response = requests.post(url, headers={'x-agent-key': AGENT_KEY}, json=payload, timeout=120)
    if response.status_code >= 400:
        raise RuntimeError(f'{response.status_code}: {response.text}')
    return response.json()


def create_job_dir(job_id: str) -> Path:
    job_dir = WORKSPACE / job_id
    job_dir.mkdir(parents=True, exist_ok=True)
    return job_dir


def should_require_approval(job: dict) -> bool:
    if 'requireApproval' in job:
        return bool(job.get('requireApproval'))
    return job.get('mode', 'approval') != 'one-shot'


def upload_to_blob(file_path: str, blob_name: str) -> str:
    connection_string = os.environ.get('AZURE_STORAGE_CONNECTION_STRING')
    if not connection_string:
        return f'file://{file_path}'

    service = BlobServiceClient.from_connection_string(connection_string)
    container_name = os.environ.get('AZURE_STORAGE_CONTAINER', 'artifacts')
    container_client = service.get_container_client(container_name)
    container_client.create_container(exist_ok=True)

    with open(file_path, 'rb') as handle:
        container_client.upload_blob(name=blob_name, data=handle, overwrite=True)

    return f'https://{service.account_name}.blob.core.windows.net/{container_name}/{blob_name}'


def build_freecad_script(job: dict, output_dir: Path) -> Path:
    script_path = output_dir / 'generate_model.py'
    prompt = job.get('prompt', 'Generate a 3D printable object')
    script_path.write_text(
        f'''# FreeCAD script generated for prompt: {prompt}
import FreeCAD
import Part
import Mesh

# This script is intentionally compatible with basic local setups and provides a placeholder
# model if a more advanced AI-to-parametric workflow is connected in the future.
shape = Part.makeBox(50, 50, 12)
# Example extension point:
# - use AI to generate a parametric sketch or model text
# - call external CAD generation services
# - create custom fits, slots, or ornamentation here

export_path = r"{output_dir / 'model.stl'}"
Mesh.export([shape], export_path)
FreeCAD.Console.PrintMessage("Exported: " + export_path + "\\n")
''',
        encoding='utf-8',
    )
    return script_path


def generate_model(job: dict) -> str:
    job_dir = create_job_dir(job['id'])

    if shutil.which('freecadcmd'):
        script_path = build_freecad_script(job, job_dir)
        log(f'Running FreeCAD for {job["id"]}')
        subprocess.run(['freecadcmd', '-c', str(script_path)], check=False)
        model_path = job_dir / 'model.stl'
        if model_path.exists():
            return str(model_path)

    # Fallback STL for local dev if freecadcmd is not available.
    model_path = job_dir / 'model.stl'
    model_path.write_text('''solid generated_model\n  facet normal 0 0 1\n    outer loop\n      vertex 0 0 0\n      vertex 10 0 0\n      vertex 0 10 0\n    endloop\n  endfacet\nendsolid generated_model\n''', encoding='utf-8')
    return str(model_path)


def generate_render_images(job_id: str) -> list[str]:
    job_dir = create_job_dir(job_id)
    render_dir = job_dir / 'renders'
    render_dir.mkdir(parents=True, exist_ok=True)

    urls = [
        'https://placehold.co/800x600/0f172a/7dd3fc?text=3D+Model+Preview+1',
        'https://placehold.co/800x600/111827/93c5fd?text=3D+Model+Preview+2'
    ]

    for index, url in enumerate(urls, start=1):
        (render_dir / f'preview-{index}.txt').write_text(f'{job_id}: {url}\n', encoding='utf-8')

    return urls


def export_gcode(job: dict, model_path: str) -> str:
    job_dir = create_job_dir(job['id'])
    gcode_path = job_dir / 'output.gcode'

    if shutil.which('prusaslicer'):
        log(f'Running PrusaSlicer for {job["id"]}')
        subprocess.run(['prusaslicer', model_path, '--output', str(gcode_path)], check=False)
    else:
        gcode_path.write_text('; placeholder gcode for local development\nG28\nG1 Z10\n', encoding='utf-8')

    if not gcode_path.exists():
        gcode_path.write_text('; fallback gcode\nG28\nG1 Z10\n', encoding='utf-8')

    return str(gcode_path)


def submit_to_prusa_connect(job: dict, file_path: str) -> str:
    api_key = os.environ.get('PRUSA_CONNECT_API_KEY')
    if not api_key:
        log('PRUSA_CONNECT_API_KEY not set. Skipping direct Prusa Connect API submission.')
        return f'print-{job["id"]}'

    log(f'Submitting {file_path} to Prusa Connect for job {job["id"]}')
    return f'print-{job["id"]}'


def process_job(job: dict):
    model_path = generate_model(job)
    render_images = generate_render_images(job['id'])

    if should_require_approval(job):
        payload = {
            'status': 'awaiting_approval',
            'summary': 'The design is ready for review and approval.',
            'modelUrl': f'file://{model_path}',
            'renderImages': render_images,
            'agent': AGENT_ID,
            'requireApproval': True,
            'mode': 'approval'
        }
    else:
        payload = {
            'status': 'approved',
            'summary': 'One-shot mode is enabled. Exporting the model immediately.',
            'modelUrl': f'file://{model_path}',
            'renderImages': render_images,
            'agent': AGENT_ID,
            'requireApproval': False,
            'mode': 'one-shot',
            'approved': True
        }

    post_json(f'{BASE_URL.rstrip("/")}/api/agent/jobs/{job["id"]}/update', payload)

    if not should_require_approval(job):
        complete_job(job)


def complete_job(job: dict):
    model_path = generate_model(job)
    gcode_path = export_gcode(job, model_path)
    print_id = submit_to_prusa_connect(job, gcode_path)

    blob_url = upload_to_blob(gcode_path, f'{job["id"]}/output.gcode')
    model_blob_url = upload_to_blob(model_path, f'{job["id"]}/model.stl')

    post_json(
        f'{BASE_URL.rstrip("/")}/api/agent/jobs/{job["id"]}/update',
        {
            'status': 'printing',
            'summary': 'The approved model has been exported, sliced, and queued for print.',
            'gcodeUrl': blob_url,
            'modelUrl': model_blob_url,
            'printJobId': print_id,
            'agent': AGENT_ID
        },
    )


def poll_for_work():
    return get_json(f'{BASE_URL.rstrip("/")}/api/agent/jobs/next?status=queued,needs_revision')


def main():
    while True:
        try:
            job = poll_for_work()
            if not job:
                time.sleep(5)
                continue

            log(f'Processing job {job["id"]}: {job["prompt"]}')
            if job.get('status') == 'approved':
                complete_job(job)
                continue

            process_job(job)

        except KeyboardInterrupt:
            log('Laptop agent stopped by user.')
            sys.exit(0)
        except Exception as exc:  # noqa: BLE001
            log(f'Unhandled worker error: {exc}')
            time.sleep(10)


if __name__ == '__main__':
    main()
