const express = require('express'); const fs = require('fs'); const path = require('path'); const app = express(); const PORT = Number(process.env.PORT || 8080); const DEFAULT_REQUIRE_APPROVALS = process.env.REQUIRE_APPROVALS !== 'false'; const DATA_DIR = path.join(__dirname, 'data'); const JOBS_FILE = path.join(DATA_DIR, 'jobs.json'); const publicDir = path.join(__dirname, 'public'); function ensureDataDir() { if (!fs.existsSync(DATA_DIR)) { fs.mkdirSync(DATA_DIR, { recursive: true }); } if (!fs.existsSync(JOBS_FILE)) { fs.writeFileSync(JOBS_FILE, '[]', 'utf8'); } } function readJobs() { ensureDataDir(); try { const raw = fs.readFileSync(JOBS_FILE, 'utf8'); const parsed = JSON.parse(raw); return Array.isArray(parsed) ? parsed : []; } catch (error) { return []; } } function writeJobs(jobs) { ensureDataDir(); fs.writeFileSync(JOBS_FILE, JSON.stringify(jobs, null, 2), 'utf8'); } function generateJobId() { return `job-${Date.now()}-${Math.random().toString(16).slice(2, 8)}`; } function createJob(prompt, opts = {}) { const now = new Date().toISOString(); const requireApproval = typeof opts.requireApproval === 'boolean' ? opts.requireApproval : DEFAULT_REQUIRE_APPROVALS; return { id: generateJobId(), prompt, status: 'queued', requireApproval, mode: opts.mode || (requireApproval ? 'approval' : 'one-shot'), createdAt: now, updatedAt: now, summary: 'Queued for the laptop agent.', modelUrl: null, renderImages: [], gcodeUrl: null, printJobId: null, approved: false, approvalHistory: [], revisions: [], agent: null }; } app.use(express.json({ limit: '20mb' })); app.use(express.static(publicDir)); app.get('/api/health', (req, res) => { res.json({ ok: true, service: '3dModelAndPrint', requireApprovals: DEFAULT_REQUIRE_APPROVALS, timestamp: new Date().toISOString() }); }); app.get('/api/jobs', (req, res) => { const jobs = readJobs().sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt)); res.json(jobs); }); app.post('/api/jobs', (req, res) => { const prompt = String(req.body?.prompt || '').trim(); if (!prompt) { return res.status(400).json({ error: 'A prompt is required.' }); } const requireApproval = typeof req.body?.requireApproval === 'boolean' ? req.body.requireApproval : (req.body?.approvalMode === 'one-shot' ? false : DEFAULT_REQUIRE_APPROVALS); const job = createJob(prompt, { requireApproval, mode: req.body?.approvalMode || (requireApproval ? 'approval' : 'one-shot') }); const jobs = readJobs(); jobs.unshift(job); writeJobs(jobs); res.status(201).json(job); }); app.get('/api/jobs/:id', (req, res) => { const jobs = readJobs(); const job = jobs.find((entry) => entry.id === req.params.id); if (!job) { return res.status(404).json({ error: 'Job not found.' }); } res.json(job); }); app.post('/api/jobs/:id/approve', (req, res) => { const jobs = readJobs(); const job = jobs.find((entry) => entry.id === req.params.id); if (!job) { return res.status(404).json({ error: 'Job not found.' }); } job.status = 'approved'; job.approved = true; job.summary = 'Approved by the user. The laptop agent can now export the model and prepare the print job.'; job.updatedAt = new Date().toISOString(); job.approvalHistory.push({ type: 'approve', user: req.body?.user || 'web-user', timestamp: job.updatedAt }); writeJobs(jobs); res.json(job); }); app.post('/api/jobs/:id/iterate', (req, res) => { const jobs = readJobs(); const job = jobs.find((entry) => entry.id === req.params.id); if (!job) { return res.status(404).json({ error: 'Job not found.' }); } const prompt = String(req.body?.prompt || '').trim(); if (!prompt) { return res.status(400).json({ error: 'A revision request is required.' }); } job.status = 'needs_revision'; job.updatedAt = new Date().toISOString(); job.summary = 'User requested a revision. The laptop agent will refine the model.'; job.revisions.push({ prompt, createdAt: job.updatedAt }); writeJobs(jobs); res.json(job); }); app.post('/api/jobs/:id/trigger-print', (req, res) => { const jobs = readJobs(); const job = jobs.find((entry) => entry.id === req.params.id); if (!job) { return res.status(404).json({ error: 'Job not found.' }); } job.status = 'exporting'; job.updatedAt = new Date().toISOString(); job.summary = 'Exporting and slicing the approved design for printing.'; writeJobs(jobs); res.json(job); }); app.get('/api/agent/jobs/next', (req, res) => { const includeStatuses = (req.query.status || 'queued,needs_revision').split(',').map((value) => value.trim()).filter(Boolean); const jobs = readJobs().sort((a, b) => new Date(a.createdAt) - new Date(b.createdAt)); const next = jobs.find((job) => includeStatuses.includes(job.status)) || null; res.json(next); }); app.post('/api/agent/jobs/:id/update', (req, res) => { const jobs = readJobs(); const job = jobs.find((entry) => entry.id === req.params.id); if (!job) { return res.status(404).json({ error: 'Job not found.' }); } const payload = req.body || {}; if (payload.status) job.status = payload.status; if (payload.summary) job.summary = payload.summary; if (payload.modelUrl) job.modelUrl = payload.modelUrl; if (Array.isArray(payload.renderImages)) job.renderImages = payload.renderImages; if (payload.gcodeUrl) job.gcodeUrl = payload.gcodeUrl; if (payload.printJobId) job.printJobId = payload.printJobId; if (payload.agent) job.agent = payload.agent; if (typeof payload.requireApproval === 'boolean') job.requireApproval = payload.requireApproval; if (payload.mode) job.mode = payload.mode; job.updatedAt = new Date().toISOString(); writeJobs(jobs); res.json(job); }); app.get('*', (req, res) => { res.sendFile(path.join(publicDir, 'index.html')); }); ensureDataDir(); app.listen(PORT, () => { console.log(`3D Model & Print server running on http://localhost:${PORT}`); });
